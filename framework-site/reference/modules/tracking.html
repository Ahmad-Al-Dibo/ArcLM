arclm.tracking core

Local experiment tracking: run manifests, metrics JSONL, environment snapshots, and reproducibility seeds. No cloud required. When to use: any training or search run you want to list, reload, or compare later.

set_reproducible_seed(seed, *, deterministic=False)

set_reproducible_seed(seed: int, *, deterministic: bool = False) -> dict
Purpose Seed Python, NumPy (if present), and torch (if present).
Returns Dict with python_seeded, numpy_seeded, torch_cpu_seeded, torch_cuda_seeded, …
Note Without torch, torch flags are False; Python seeding still runs.
# Works on core install
from arclm.tracking import set_reproducible_seed
report = set_reproducible_seed(42)
print(report["python_seeded"], report["torch_cpu_seeded"])
# True False   (without torch)

# With torch installed
report = set_reproducible_seed(42, deterministic=False)
print(report["torch_cpu_seeded"])  # True

create_run(config: RunConfig)

create_run(config: RunConfig) -> RunTracker
Purpose Create a run directory under the storage root and return a tracker.
Params RunConfig(name, operation, training_method, storage=RunStorageConfig(...), …)
from arclm.tracking import (
    create_run, RunConfig, RunOperation, RunStorageConfig,
    load_run_manifest, list_runs, validate_run, compare_runs,
    RunStatus,
)
from pathlib import Path

tracker = create_run(RunConfig(
    name="demo",
    operation=RunOperation.TRAINING,
    storage=RunStorageConfig(root_directory=Path("runs-demo"), fsync=False),
))
tracker.start()
tracker.record_metric(metrics={"loss": 2.0}, step=1)
tracker.record_metric(metrics={"loss": 1.0}, step=2)
tracker.complete({"final_loss": 1.0})
tracker.close()

manifest = load_run_manifest(tracker.run_dir)
print(manifest.status)  # completed

runs = list_runs("runs-demo", status=RunStatus.COMPLETED)
print(len(runs), runs[0].run_id)

info = validate_run(tracker.run_dir)
print(info.metric_count)

cmp = compare_runs([tracker.run_dir], metric_name="loss")
print(cmp.metric_summary)
