arclm.training needs [train]

Native training loop, unified pipeline, regularization, and checkpoint helpers.

class TrainingConfig

TrainingConfig(**dataclass fields)
Purpose Small validated config for the modern Trainer API (separate from arclm.configs.Config).
Key fields epochs=3, batch_size=8, learning_rate, weight_decay, precision="fp32" (fp16|bf16), optimizer="adamw", scheduler="cosine", max_grad_norm, output_dir, …
Errors TrainingConfigError on invalid values (epochs < 1, bad precision, …).
from arclm.training import TrainingConfig
from arclm.exceptions import TrainingConfigError

cfg = TrainingConfig(epochs=5, batch_size=4, precision="fp32")
print(cfg.to_dict())
restored = TrainingConfig.from_dict(cfg.to_dict())
assert restored == cfg

# TrainingConfig(epochs=0)  -> TrainingConfigError

class Trainer

Trainer(model, config=None, callbacks=None)
Purpose Train an ArcLM-style model on token loaders; save/load checkpoints.
Methods train_dataset(train_loader, val_loader=None), save(...), load(...), add_callback

train_model(...)

train_model(mode, data, output, checkpoint=None, config=None, …)
Purpose One-call high-level training (pre_training / fine_tuning / instruction_tuning).
Returns Result object with history (e.g. train_losses) and model_path.
# Smoke-scale path used in examples/04_training
from arclm.configs import create_config
from arclm.training import train_model

config = create_config(embed_dim=32, block_size=16, num_epochs=1, device="cpu")
result = train_model(
    mode="pre_training",
    data="data.txt",
    output="models/tiny.pth",
    config=config,
)
print(result.history.get("train_losses", [])[-1])

Checkpoint compatibility helpers

checkpoint_is_compatible_for_continue_training(checkpoint, config, vocab_size, tokenizer=None) -> bool
Purpose Strict resume check: vocab, shape, tokenizer, and training hyperparameters must match. Empty/missing config → False.
checkpoint_is_compatible_for_tuning(checkpoint, config, vocab_size, tokenizer=None) -> bool
Purpose Fine-tune check: shape + tokenizer (learning rate may differ). Empty config → False.
load_compatible_checkpoint(trainer, config, vocab_size, tokenizer=None) -> bool
